508 template <class RealExpression, class ImagExpression>
509 constexpr void assignSplitComplexExpression (const RealExpression& re, const ImagExpression& im)
510 {
511 VCTR_ASSERT (re.size() == size());
512 VCTR_ASSERT (im.size() == size());
513
514 if (! std::is_constant_evaluated())
515 {
516 if constexpr (has::getNeon<RealExpression> && has::getNeon<ImagExpression>)
517 {
518 assignSplitComplexExpressionNeon (re, im);
519 return;
520 }
521
522 if constexpr (has::getAVX<RealExpression> && has::getAVX<ImagExpression>)
523 {
524 if (Config::supportedCPUInstructionSets.fma)
525 {
526 assignSplitComplexExpressionFMA (re, im);
527 return;
528 }
529 }
530
531 if constexpr (has::getSSE<RealExpression> && has::getSSE<ImagExpression>)
532 {
533 if (Config::supportedCPUInstructionSets.sse4_1)
534 {
535 assignSplitComplexExpressionSSE4_1 (re, im);
536 return;
537 }
538 }
539 }
540
541 assignSplitComplexExpressionScalar (re, im, 0);
542 }

545 template <class RealExpression, class ImagExpression>
546 constexpr void assignSplitComplexExpressionScalar (const RealExpression& re, const ImagExpression& im, size_t startIdx)
547 {
548 auto* r = realParts.data();
549 auto* i = imagParts.data();
550 const auto n = size();
551
552 for (auto idx = startIdx; idx < n; ++idx)
553 {
554 const value_type realValue = re[idx];
555 const value_type imagValue = im[idx];
556 r[idx] = realValue;
557 i[idx] = imagValue;
558 }
559 }
560
562 template <class RealExpression, class ImagExpression>
563 void assignSplitComplexExpressionNeon (const RealExpression& re, const ImagExpression& im)
564 requires Config::archARM
565 {
566 constexpr auto inc = NeonRegister<value_type>::numElements;
567 const auto nSIMD = detail::previousMultipleOf<inc> (size());
568
569 re.prepareNeonEvaluation();
570 im.prepareNeonEvaluation();
571 auto* r = realParts.data();
572 auto* i = imagParts.data();
573
574 size_t idx = 0;
575 for (; idx < nSIMD; idx += inc)
576 {
577 const auto realValues = re.getNeon (idx);
578 const auto imagValues = im.getNeon (idx);
579 realValues.store (r + idx);
580 imagValues.store (i + idx);
581 }
582
583 assignSplitComplexExpressionScalar (re, im, idx);
584 }

589 template <class RealExpression, class ImagExpression>
590 VCTR_TARGET ("fma")
591 void assignSplitComplexExpressionFMA (const RealExpression& re, const ImagExpression& im)
592 requires Config::archX64
593 {
594 constexpr auto inc = AVXRegister<value_type>::numElements;
595 const auto nSIMD = detail::previousMultipleOf<inc> (size());
596
597 re.prepareAVXEvaluation();
598 im.prepareAVXEvaluation();
599 auto* r = realParts.data();
600 auto* i = imagParts.data();
601
602 size_t idx = 0;
603 for (; idx < nSIMD; idx += inc)
604 {
605 const auto realValues = re.getAVX (idx);
606 const auto imagValues = im.getAVX (idx);
607 realValues.storeUnaligned (r + idx);
608 imagValues.storeUnaligned (i + idx);
609 }
610
611 assignSplitComplexExpressionScalar (re, im, idx);
612 }

617 template <class RealExpression, class ImagExpression>
