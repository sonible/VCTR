26template <size_t extent, class SrcAType, class SrcBType>
27requires is::realFloatNumber<ValueType<SrcAType>> && std::same_as<ValueType<SrcAType>, ValueType<SrcBType>>
28class RealImagToComplex : public ExpressionTemplateBase
29{
30public:
31 using value_type = std::complex<ValueType<SrcAType>>;
32
33 VCTR_COMMON_BINARY_VEC_VEC_EXPRESSION_MEMBERS (RealImagToComplex, real, imag)
34
35 VCTR_FORCEDINLINE constexpr value_type operator[] (size_t i) const
36 {
37 return { real[i], imag[i] };
38 }
39
40 //==============================================================================
41 // Platform Vector Operation Implementation. The real sources can't be evaluated into the complex destination, so the
42 // concepts only allow containers as sources. Expression::Accelerate / Expression::IPP refer to the real source type,
43 // the functions live in the complex one.
44 VCTR_FORCEDINLINE const value_type* evalNextVectorOpInExpressionChain (value_type* dst) const
45 requires is::suitableForAccelerateComplexFloatBinaryVectorOp<SrcAType, SrcBType, value_type, detail::dontPreferIfIppAndAccelerateAreAvailable>
46 {
47 platform_vec_ops::AppleAccelerate<value_type>::realImagToComplex (real.data(), imag.data(), dst, size());
48 return dst;
49 }
50
51 VCTR_FORCEDINLINE const value_type* evalNextVectorOpInExpressionChain (value_type* dst) const
52 requires is::suitableForIppComplexFloatBinaryVectorOp<SrcAType, SrcBType, value_type, detail::preferIfIppAndAccelerateAreAvailable>
53 {
54 platform_vec_ops::IntelIPP<value_type>::realImagToComplex (real.data(), imag.data(), dst, sizeToInt (size()));
55 return dst;
56 }
57};
58
59} // namespace vctr::expressions

69template <is::anyVctrOrExpression RealType, is::anyVctrOrExpression ImagType>
70constexpr auto realImagToComplex (RealType&& real, ImagType&& imag)
71{
72 assertCommonSize (real, imag);
73 constexpr auto extent = getCommonExtent<RealType, ImagType>();
74 return expressions::RealImagToComplex<extent, RealType, ImagType> (std::forward<RealType> (real), std::forward<ImagType> (imag));
75}
