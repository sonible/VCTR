62template <is::anySplitComplexVectorOrExpression SrcAType, is::anySplitComplexVectorOrExpression SrcBType>
63constexpr auto operator* (SrcAType&& srcA, SrcBType&& srcB)
64{
65 // (a + bi) * (c + di) = (ac - bd) + (ad + bc)i
66 // The sub-expressions are built inline rather than stored in local variables first: expressions hold lvalue
67 // sources by reference, so an expression built from locals would dangle after returning.
68 return makeSplitComplexExpression (srcA.real() * srcB.real() - srcA.imag() * srcB.imag(),
69 srcA.real() * srcB.imag() + srcA.imag() * srcB.real());
70}

139template <is::anySplitComplexVectorOrExpression SrcAType, is::anySplitComplexVectorOrExpression SrcBType>
140constexpr auto operator/ (SrcAType&& srcA, SrcBType&& srcB)
141{
142 // (a + bi) / (c + di) = ((ac + bd) + (bc - ad)i) / (c² + d²)
143 // The denominator is built inline for both parts on purpose, see operator*.
144 return makeSplitComplexExpression ((srcA.real() * srcB.real() + srcA.imag() * srcB.imag()) / (srcB.real() * srcB.real() + srcB.imag() * srcB.imag()),
145 (srcA.imag() * srcB.real() - srcA.real() * srcB.imag()) / (srcB.real() * srcB.real() + srcB.imag() * srcB.imag()));
146}
